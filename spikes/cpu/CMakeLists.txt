cmake_minimum_required(VERSION 3.24)
project(bb_cpu_feasibility LANGUAGES C)
include(CTest)
include(FetchContent)
find_package(Threads REQUIRED)
find_program(BB_PKG_CONFIG NAMES pkg-config REQUIRED)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
    message(FATAL_ERROR "This M1 experiment requires native arm64 macOS; use the root harness for other hosts.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_ARCH x86 CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
# Upstream otherwise defaults to a universal build. Limit the dependency to the
# same native architecture as this experiment and restore the environment after it.
set(bb_saved_archflags "$ENV{ARCHFLAGS}")
if(DEFINED ENV{ARCHFLAGS})
    set(bb_had_archflags TRUE)
endif()
set(ENV{ARCHFLAGS} "-arch arm64")
FetchContent_Declare(unicorn
    URL https://codeload.github.com/unicorn-engine/unicorn/tar.gz/8028ec436f2d9376525352dd38ed9ed6b9f6be10
    URL_HASH SHA256=9d2a2003f0307d0fc239af439654564fe6ef17b67f7c5a39b45fad4c69d6aec2
    DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
# QEMU's ARM cache helpers use GNU inline assembly. Keep extensions scoped to
# the dependency; the probe itself remains strict C11.
set(CMAKE_C_EXTENSIONS ON)
FetchContent_MakeAvailable(unicorn)
set(CMAKE_C_EXTENSIONS OFF)
if(bb_had_archflags)
    set(ENV{ARCHFLAGS} "${bb_saved_archflags}")
else()
    unset(ENV{ARCHFLAGS})
endif()
file(READ "${CMAKE_BINARY_DIR}/config-host.h" bb_unicorn_host_config)
if(NOT bb_unicorn_host_config MATCHES "#define HAVE_PTHREAD_JIT_PROTECT 1")
    message(FATAL_ERROR "Unicorn did not detect Apple's JIT protection API; inspect dependency configure output.")
endif()
add_library(bb-cpu-probe STATIC probe.c)
target_include_directories(bb-cpu-probe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_compile_options(bb-cpu-probe PRIVATE -Wall -Wextra -Werror -fno-omit-frame-pointer)
target_link_libraries(bb-cpu-probe PUBLIC unicorn Threads::Threads)
add_executable(cpu-probe-smoke smoke.c)
target_compile_options(cpu-probe-smoke PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(cpu-probe-smoke PRIVATE bb-cpu-probe)
add_test(NAME cpu-probe-smoke COMMAND cpu-probe-smoke)
set_tests_properties(cpu-probe-smoke PROPERTIES TIMEOUT 30)
