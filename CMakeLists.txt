cmake_minimum_required(VERSION 3.24)
project(bb_cpu_harness LANGUAGES C)

include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    message(FATAL_ERROR "The runtime requires GCC or Clang; use MSYS2 CLANG64 on Windows.")
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Darwin|Linux|Windows)$")
    message(FATAL_ERROR "Unsupported host: ${CMAKE_SYSTEM_NAME}")
endif()
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" BB_PROCESSOR)
if(APPLE AND CMAKE_OSX_ARCHITECTURES AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL CMAKE_SYSTEM_PROCESSOR)
    message(FATAL_ERROR "The M0 CMake harness uses one host architecture per build tree; universal/cross builds are not supported yet.")
endif()
if(BB_PROCESSOR MATCHES "^(x86_64|amd64)$")
    set(BB_X86_HOST TRUE)
    set(BB_DEFAULT_BACKEND native-x86)
elseif(BB_PROCESSOR MATCHES "^(arm64|aarch64)$")
    set(BB_DEFAULT_BACKEND unavailable)
else()
    message(FATAL_ERROR "Unsupported CPU: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(BB_EXECUTION_BACKEND "${BB_DEFAULT_BACKEND}" CACHE STRING "native-x86 or unavailable (host contracts only)")
set_property(CACHE BB_EXECUTION_BACKEND PROPERTY STRINGS native-x86 unavailable)
if(NOT BB_EXECUTION_BACKEND MATCHES "^(native-x86|unavailable)$")
    message(FATAL_ERROR "Unknown BB_EXECUTION_BACKEND=${BB_EXECUTION_BACKEND}; no ARM translator is integrated yet.")
endif()
if(BB_EXECUTION_BACKEND STREQUAL "native-x86" AND NOT BB_X86_HOST)
    message(FATAL_ERROR "native-x86 requires an x86-64 host; ARM guest execution is an M1 dependency.")
endif()
set(BB_NATIVE_LOADER_DEFAULT OFF)
if(BB_X86_HOST AND NOT APPLE AND BB_EXECUTION_BACKEND STREQUAL "native-x86")
    set(BB_NATIVE_LOADER_DEFAULT ON)
endif()
option(BB_BUILD_NATIVE_LOADER "Build the CPU-only PS4 loader and native runtime tests" ${BB_NATIVE_LOADER_DEFAULT})
if(BB_BUILD_NATIVE_LOADER AND (APPLE OR NOT BB_X86_HOST OR NOT BB_EXECUTION_BACKEND STREQUAL "native-x86"))
    message(FATAL_ERROR "The native loader currently supports x86-64 Linux/Windows only. Use the host-contract harness on macOS.")
endif()

function(bb_runtime_target target)
    target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/src")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-omit-frame-pointer)
    # Threads::Threads can be empty on modern glibc. Match build.sh's -pthread:
    # it also enables POSIX declarations needed by the strict-C11 runtime headers.
    target_compile_options(${target} PUBLIC -pthread)
    target_link_options(${target} PUBLIC -pthread)
    target_link_libraries(${target} PUBLIC Threads::Threads)
    if(NOT BB_X86_HOST)
        target_compile_definitions(${target} PUBLIC BB_HOST_CONTRACTS_ONLY)
    endif()
endfunction()

set(BB_CORE_SOURCES src/runtime.c src/runtime_execution.c src/runtime_subsystems.c src/runtime_content.c src/runtime_time.c src/runtime_rtc.c)
add_library(bb-runtime-core STATIC ${BB_CORE_SOURCES})
bb_runtime_target(bb-runtime-core)
target_compile_definitions(bb-runtime-core PRIVATE BB_RUNTIME_CORE_ONLY)
if(BB_EXECUTION_BACKEND STREQUAL "unavailable")
    target_compile_definitions(bb-runtime-core PRIVATE BB_EXECUTION_UNAVAILABLE)
endif()
if(WIN32)
    target_sources(bb-runtime-core PRIVATE src/compat_win.c)
    target_link_libraries(bb-runtime-core PUBLIC bcrypt)
endif()

if(BUILD_TESTING)
    add_executable(execution-test tests/test_execution.c)
    bb_runtime_target(execution-test)
    target_compile_options(execution-test PRIVATE -UNDEBUG)
    target_link_libraries(execution-test PRIVATE bb-runtime-core)
    add_test(NAME execution-interface COMMAND execution-test)
    if(BB_EXECUTION_BACKEND STREQUAL "unavailable")
        add_test(NAME execution-unavailable-diagnostic
            COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/expect_exit.py"
                "$<TARGET_FILE:execution-test>" execution-unavailable)
    endif()
    add_executable(runtime-core-test tests/test_runtime_core.c)
    bb_runtime_target(runtime-core-test)
    target_compile_options(runtime-core-test PRIVATE -UNDEBUG)
    target_link_libraries(runtime-core-test PRIVATE bb-runtime-core)
    add_executable(content-test tests/test_content.c)
    bb_runtime_target(content-test)
    target_compile_options(content-test PRIVATE -UNDEBUG)
    target_link_libraries(content-test PRIVATE bb-runtime-core)
    add_test(NAME runtime-core COMMAND runtime-core-test)
    foreach(case guard-recursion stack-failure bad-tls)
        add_test(NAME runtime-core-${case}
            COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/expect_exit.py"
                "$<TARGET_FILE:runtime-core-test>" "${case}")
    endforeach()
    add_test(NAME content-contracts COMMAND "${Python3_EXECUTABLE}" -m unittest discover
        -s "${PROJECT_SOURCE_DIR}/tests" -p test_content.py -v)
    set_tests_properties(content-contracts PROPERTIES ENVIRONMENT
        "BB_TEST_BIN_DIR=$<TARGET_FILE_DIR:content-test>;BB_REQUIRED_NATIVE_TESTS=content-test")
    set(BB_PYTHON_SUITES prepare link_libc patches build_harness)
    if(NOT WIN32)
        list(APPEND BB_PYTHON_SUITES mods)
    endif()
    foreach(suite IN LISTS BB_PYTHON_SUITES)
        add_test(NAME python-${suite} COMMAND "${Python3_EXECUTABLE}" -m unittest discover
            -s "${PROJECT_SOURCE_DIR}/tests" -p "test_${suite}.py" -v)
    endforeach()
    set_tests_properties(python-build_harness PROPERTIES ENVIRONMENT
        "CMAKE_GENERATOR=${CMAKE_GENERATOR};CC=${CMAKE_C_COMPILER}")
endif()

if(APPLE)
    add_library(bb-guest-memory STATIC src/platform/guest_memory_darwin.c)
    bb_runtime_target(bb-guest-memory)
    if(BUILD_TESTING)
        add_executable(guest-memory-test tests/test_guest_memory.c)
        bb_runtime_target(guest-memory-test)
        target_compile_options(guest-memory-test PRIVATE -UNDEBUG)
        target_link_libraries(guest-memory-test PRIVATE bb-guest-memory)
        add_test(NAME darwin-guest-memory COMMAND guest-memory-test)
    endif()
endif()

if(BB_BUILD_NATIVE_LOADER)
    set(BB_NATIVE_SOURCES ${BB_CORE_SOURCES} src/runtime_mutex.c src/runtime_rwlock.c src/runtime_sema.c
        src/runtime_memory.c src/runtime_kernel.c src/runtime_thread.c src/runtime_file.c
        src/runtime_fiber.c src/runtime_services.c src/runtime_savedata.c src/cpu_only_services.c)
    add_library(bb-runtime-native STATIC ${BB_NATIVE_SOURCES})
    bb_runtime_target(bb-runtime-native)
    target_compile_definitions(bb-runtime-native PRIVATE BB_RUNTIME_CPU_ONLY)
    if(WIN32)
        target_sources(bb-runtime-native PRIVATE src/compat_win.c)
        target_link_libraries(bb-runtime-native PUBLIC onecore bcrypt ws2_32)
    else()
        target_link_libraries(bb-runtime-native PUBLIC m ${CMAKE_DL_LIBS})
    endif()
    add_executable(bb-probe src/probe.c)
    bb_runtime_target(bb-probe)
    target_link_libraries(bb-probe PRIVATE bb-runtime-native)
    if(WIN32)
        target_link_options(bb-probe PRIVATE -Wl,--disable-high-entropy-va "SHELL:-Xlinker --stack -Xlinker 16777216,16777216")
    else()
        target_link_options(bb-probe PRIVATE -no-pie -rdynamic)
    endif()
    if(BUILD_TESTING)
        set(BB_NATIVE_TESTS "runtime-test|tests/test_runtime.c" "sema-test|tests/test_sema.c")
        if(NOT WIN32)
            list(APPEND BB_NATIVE_TESTS "file-mods-test|tests/test_file_mods.c")
        endif()
        foreach(pair IN LISTS BB_NATIVE_TESTS)
            string(REPLACE "|" ";" parts "${pair}")
            list(GET parts 0 target)
            list(GET parts 1 source)
            add_executable(${target} ${source})
            bb_runtime_target(${target})
            target_compile_options(${target} PRIVATE -UNDEBUG)
            if(NOT target STREQUAL "file-mods-test")
                target_link_libraries(${target} PRIVATE bb-runtime-native)
            endif()
        endforeach()
        if(WIN32)
            # This fixture exercises the same fixed PS4 address space as bb-probe.
            target_link_options(runtime-test PRIVATE -Wl,--disable-high-entropy-va)
        endif()
        if(NOT WIN32)
            add_test(NAME file-mods COMMAND file-mods-test)
        endif()
        foreach(suite probe runtime sema)
            add_test(NAME native-${suite} COMMAND "${Python3_EXECUTABLE}" -m unittest discover
                -s "${PROJECT_SOURCE_DIR}/tests" -p "test_${suite}.py" -v)
            set_tests_properties(native-${suite} PROPERTIES ENVIRONMENT
                "BB_TEST_BIN_DIR=$<TARGET_FILE_DIR:bb-probe>;BB_REQUIRED_NATIVE_TESTS=bb-probe,runtime-test,sema-test")
        endforeach()
    endif()
endif()
if(BUILD_TESTING)
    get_property(bb_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${bb_tests} PROPERTIES TIMEOUT 60)
endif()
message(STATUS "Host: ${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}; execution backend: ${BB_EXECUTION_BACKEND}")
message(STATUS "Native loader: ${BB_BUILD_NATIVE_LOADER}; no graphics/network dependencies in this build")
